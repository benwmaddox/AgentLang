word customer.discounted-balance(customer: Customer) -> Money {
    effects none
    doc "For raw kind exactly premium, returns 90% of the signed balance, truncating only the final fractional minor unit toward zero; other kinds keep their balance."
    let balance = customer::balance(customer);
    if ::equals(customer::kind(customer), "premium") {
        let minor-units = Money::value(balance);
        let quotient = ::divide(minor-units, 10);
        let quotient-part = ::multiply(quotient, 9);
        let remainder = ::subtract(minor-units, ::multiply(quotient, 10));
        let remainder-part = ::divide(::multiply(remainder, 9), 10);
        Money::new(::add(quotient-part, remainder-part))
    } else {
        balance
    }
}
