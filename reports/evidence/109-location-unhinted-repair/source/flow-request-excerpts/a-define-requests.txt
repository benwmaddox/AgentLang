word customer.discounted-balance(customer: Customer) -> Money {
    effects none
    doc "For customers whose raw kind is exactly premium, returns 90 percent of the signed balance with the final fractional minor unit truncated toward zero; other kinds keep their balance."
    let balance = customer::balance(customer);
    if customer::premium?(customer) {
        let minor-units = Money::value(balance);
        let tens = ::divide(minor-units, 10);
        let remainder = ::subtract(minor-units, ::multiply(tens, 10));
        let whole-percent = ::multiply(tens, 9);
        let remainder-percent = ::divide(::multiply(remainder, 9), 10);
        Money::new(::add(whole-percent, remainder-percent))
    } else {
        balance
    }
}
